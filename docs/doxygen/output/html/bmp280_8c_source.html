116static bool read_hum_calibration_data(BMP280_HandleTypedef *dev) {
117 uint16_t h4, h5;
118
119 if (!read_data(dev, 0xa1, &dev->dig_H1, 1)
120 && read_register16(dev, 0xe1, (uint16_t *) &dev->dig_H2)
121 && !read_data(dev, 0xe3, &dev->dig_H3, 1)
122 && read_register16(dev, 0xe4, &h4)
123 && read_register16(dev, 0xe5, &h5)
124 && !read_data(dev, 0xe7, (uint8_t *) &dev->dig_H6, 1)) {
125 dev->dig_H4 = (h4 & 0x00ff) << 4 | (h4 & 0x0f00) >> 8;
126 dev->dig_H5 = h5 >> 4;
127
128 return true;
129 }

131 return false;

146bool bmp280_init(BMP280_HandleTypedef *dev, bmp280_params_t *params) {
147
148 if (dev->addr != BMP280_I2C_ADDRESS_0
149 && dev->addr != BMP280_I2C_ADDRESS_1) {
150
151 return false;
152 }
153
154 if (read_data(dev, BMP280_REG_ID, &dev->id, 1)) {
155 return false;
156 }
157
158 if (dev->id != BMP280_CHIP_ID && dev->id != BME280_CHIP_ID) {
159
160 return false;
161 }
162
163 // Soft reset.
164 if (write_register8(dev, BMP280_REG_RESET, BMP280_RESET_VALUE)) {
165 return false;
166 }
167
168 // Wait until finished copying over the NVP data.
169 while (1) {
170 uint8_t status;
171 if (!read_data(dev, BMP280_REG_STATUS, &status, 1)
172 && (status & 1) == 0)
173 break;
174 }
175
176 if (!read_calibration_data(dev)) {
177 return false;
178 }
179
180 if (dev->id == BME280_CHIP_ID && !read_hum_calibration_data(dev)) {
181 return false;
182 }
183
184 uint8_t config = (params->standby << 5) | (params->filter << 2);
185 if (write_register8(dev, BMP280_REG_CONFIG, config)) {
186 return false;
187 }
188
189 if (params->mode == BMP280_MODE_FORCED) {
190 params->mode = BMP280_MODE_SLEEP; // initial mode for forced is sleep
191 }
192
193 uint8_t ctrl = (params->oversampling_temperature << 5)
194 | (params->oversampling_pressure << 2) | (params->mode);
195
196 if (dev->id == BME280_CHIP_ID) {
197 // Write crtl hum reg first, only active after write to BMP280_REG_CTRL.
198 uint8_t ctrl_hum = params->oversampling_humidity;
199 if (write_register8(dev, BMP280_REG_CTRL_HUM, ctrl_hum)) {
200 return false;
201 }
202 }
203
204 if (write_register8(dev, BMP280_REG_CTRL, ctrl)) {
205 return false;
206 }
207
208 return true;
209}

211bool bmp280_force_measurement(BMP280_HandleTypedef *dev) {
212 uint8_t ctrl;

214 return false;

216 ctrl |= BMP280_MODE_FORCED;
217 if (write_register8(dev, BMP280_REG_CTRL, ctrl)) {

238static inline int32_t compensate_temperature(BMP280_HandleTypedef *dev, int32_t adc_temp,
239 int32_t *fine_temp) {
240 int32_t var1, var2;
241
242 var1 = ((((adc_temp >> 3) - ((int32_t) dev->dig_T1 << 1)))
243 * (int32_t) dev->dig_T2) >> 11;
244 var2 = (((((adc_temp >> 4) - (int32_t) dev->dig_T1)
245 * ((adc_temp >> 4) - (int32_t) dev->dig_T1)) >> 12)
246 * (int32_t) dev->dig_T3) >> 14;
247
248 *fine_temp = var1 + var2;
249 return (*fine_temp * 5 + 128) >> 8;
250}

257static inline uint32_t compensate_pressure(BMP280_HandleTypedef *dev, int32_t adc_press,
258 int32_t fine_temp) {
259 int64_t var1, var2, p;
260
261 var1 = (int64_t) fine_temp - 128000;
262 var2 = var1 * var1 * (int64_t) dev->dig_P6;
263 var2 = var2 + ((var1 * (int64_t) dev->dig_P5) << 17);
264 var2 = var2 + (((int64_t) dev->dig_P4) << 35);
265 var1 = ((var1 * var1 * (int64_t) dev->dig_P3) >> 8)
266 + ((var1 * (int64_t) dev->dig_P2) << 12);
267 var1 = (((int64_t) 1 << 47) + var1) * ((int64_t) dev->dig_P1) >> 33;
268
269 if (var1 == 0) {
270 return 0; // avoid exception caused by division by zero
271 }
272
273 p = 1048576 - adc_press;
274 p = (((p << 31) - var2) * 3125) / var1;
275 var1 = ((int64_t) dev->dig_P9 * (p >> 13) * (p >> 13)) >> 25;
276 var2 = ((int64_t) dev->dig_P8 * p) >> 19;
277
278 p = ((p + var1 + var2) >> 8) + ((int64_t) dev->dig_P7 << 4);
279 return p;
280}

287static inline uint32_t compensate_humidity(BMP280_HandleTypedef *dev, int32_t adc_hum,
288 int32_t fine_temp) {
289 int32_t v_x1_u32r;
290
291 v_x1_u32r = fine_temp - (int32_t) 76800;
292 v_x1_u32r = ((((adc_hum << 14) - ((int32_t) dev->dig_H4 << 20)
293 - ((int32_t) dev->dig_H5 * v_x1_u32r)) + (int32_t) 16384) >> 15)
294 * (((((((v_x1_u32r * (int32_t) dev->dig_H6) >> 10)
295 * (((v_x1_u32r * (int32_t) dev->dig_H3) >> 11)
296 + (int32_t) 32768)) >> 10) + (int32_t) 2097152)
297 * (int32_t) dev->dig_H2 + 8192) >> 14);
298 v_x1_u32r = v_x1_u32r
299 - (((((v_x1_u32r >> 15) * (v_x1_u32r >> 15)) >> 7)
300 * (int32_t) dev->dig_H1) >> 4);
301 v_x1_u32r = v_x1_u32r < 0 ? 0 : v_x1_u32r;
302 v_x1_u32r = v_x1_u32r > 419430400 ? 419430400 : v_x1_u32r;
303 return v_x1_u32r >> 12;
304}

306bool bmp280_read_fixed(BMP280_HandleTypedef *dev, int32_t *temperature, uint32_t *pressure,
307 uint32_t *humidity) {
308 int32_t adc_pressure;
309 int32_t adc_temp;
310 uint8_t data[8];
311
312 // Only the BME280 supports reading the humidity.
313 if (dev->id != BME280_CHIP_ID) {
314 if (humidity)
315 *humidity = 0;
316 humidity = NULL;
317 }
318
319 // Need to read in one sequence to ensure they match.
320 size_t size = humidity ? 8 : 6;
321 if (read_data(dev, 0xf7, data, size)) {
322 return false;
323 }
324
325 adc_pressure = data[0] << 12 | data[1] << 4 | data[2] >> 4;
326 adc_temp = data[3] << 12 | data[4] << 4 | data[5] >> 4;
327
328 int32_t fine_temp;
329 *temperature = compensate_temperature(dev, adc_temp, &fine_temp);
330 *pressure = compensate_pressure(dev, adc_pressure, fine_temp);
331
332 if (humidity) {
333 int32_t adc_humidity = data[6] << 8 | data[7];
334 *humidity = compensate_humidity(dev, adc_humidity, fine_temp);
335 }
336
337 return true;
338}

340bool bmp280_read_float(BMP280_HandleTypedef *dev, float *temperature, float *pressure,
341 float *humidity) {
342 int32_t fixed_temperature;
343 uint32_t fixed_pressure;
344 uint32_t fixed_humidity;
345 if (bmp280_read_fixed(dev, &fixed_temperature, &fixed_pressure,
346 humidity ? &fixed_humidity : NULL)) {
347 *temperature = (float) fixed_temperature / 100;
348 *pressure = (float) fixed_pressure / 256;
349 if (humidity)
350 *humidity = (float) fixed_humidity / 1024;
351 return true;
352 }
353
354 return false;
