17static uint8_t radio_normalize_backend(uint8_t modulation)
18{
19 if ((modulation == RADIO_LIB_MODULATION_LORA) ||
20 (modulation == RADIO_LIB_MODULATION_FSK) ||
21 (modulation == RADIO_LIB_MODULATION_OOK))
22 {
23 return modulation;
24 }
25
26 return RADIO_LIB_MODULATION_LORA;
27}

39void radio_default_fsk_cfg(radio_fsk_cfg_t *cfg)
40{
41 if (cfg == NULL)
42 {
43 return;
44 }
45
46 memset(cfg, 0, sizeof(*cfg));
47 cfg->frequency_hz = 868300000UL;
48 cfg->bitrate_bps = 4800UL;
49 cfg->rx_bandwidth = RADIO_LORA_BW_125_KHZ;
50 cfg->shaping = RADIO_FSK_SHAPING_GFSK;
51 cfg->filter = RADIO_FSK_FILTER_BT_05;
52 cfg->tx_power_dbm = 14;
53 cfg->preamble_len = 8U;
54 cfg->sync_word_len = 2U;
55 cfg->sync_word = 0x00002DD4UL;
56 cfg->address_filter = RADIO_ADDRESS_FILTER_OFF;
57 cfg->crc_type = RADIO_PACKET_CRC_CCITT;
58 cfg->data_whitening = true;
59}
