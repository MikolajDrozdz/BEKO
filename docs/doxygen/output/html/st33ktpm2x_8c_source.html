154static st33ktpm2x_status_t st33_i2c_read(st33ktpm2x_t *ctx,
155 uint8_t reg,
156 uint8_t *data,
157 uint16_t len)
158{
159 HAL_StatusTypeDef hal_st = HAL_ERROR;
160 uint8_t attempt;
161
162 if ((ctx == NULL) || (data == NULL) || (len == 0U))
163 {
164 return ST33KTPM2X_EINVAL;
165 }
166
167 for (attempt = 0U; attempt < ST33_I2C_MAX_RETRIES; attempt++)
168 {
169 hal_st = HAL_I2C_Master_Transmit(ctx->cfg.hi2c,
170 (uint16_t)(ctx->cfg.i2c_addr_7bit << 1),
171 &reg,
172 1U,
173 ctx->cfg.io_timeout_ms);
174 if (hal_st == HAL_OK)
175 {
176 app_delay_ms(1U);
177 hal_st = HAL_I2C_Master_Receive(ctx->cfg.hi2c,
178 (uint16_t)(ctx->cfg.i2c_addr_7bit << 1),
179 data,
180 len,
181 ctx->cfg.io_timeout_ms);
182 }
183 if (hal_st == HAL_OK)
184 {
185 app_delay_ms(1U);
186 return ST33KTPM2X_OK;
187 }
188 app_delay_ms(1U);
189 }
190
191 if (hal_st == HAL_TIMEOUT)
192 {
193 return ST33KTPM2X_ETIMEOUT;
194 }
195 return ST33KTPM2X_EHAL;
196}

199static st33ktpm2x_status_t st33_i2c_write(st33ktpm2x_t *ctx,
200 uint8_t reg,
201 const uint8_t *data,
202 uint16_t len)
203{
204 HAL_StatusTypeDef hal_st = HAL_ERROR;
205 uint8_t attempt;
206
207 if ((ctx == NULL) || (data == NULL) || (len == 0U))
208 {
209 return ST33KTPM2X_EINVAL;
210 }
211 if (len > ST33KTPM2X_CMD_MAX_BYTES)
212 {
213 return ST33KTPM2X_EOVERFLOW;
214 }
215
216 s_st33_i2c_tx[0] = reg;
217 memcpy(&s_st33_i2c_tx[1], data, len);
218
219 for (attempt = 0U; attempt < ST33_I2C_MAX_RETRIES; attempt++)
220 {
221 hal_st = HAL_I2C_Master_Transmit(ctx->cfg.hi2c,
222 (uint16_t)(ctx->cfg.i2c_addr_7bit << 1),
223 s_st33_i2c_tx,
224 (uint16_t)(len + 1U),
225 ctx->cfg.io_timeout_ms);
226 if (hal_st == HAL_OK)
227 {
228 app_delay_ms(1U);
229 return ST33KTPM2X_OK;
230 }
231 app_delay_ms(1U);
232 }
233
234 if (hal_st == HAL_TIMEOUT)
235 {
236 return ST33KTPM2X_ETIMEOUT;
237 }
238 return ST33KTPM2X_EHAL;
239}

390static st33ktpm2x_status_t st33_wait_burst(st33ktpm2x_t *ctx, uint16_t *burst_out, uint8_t require_data_avail)
391{
392 uint32_t start_ms;
393 uint8_t sts = 0U;
394 uint16_t burst = 0U;
395 uint8_t last_sts = 0U;
396 uint16_t last_burst = 0U;
397 st33ktpm2x_status_t rc;
398 st33ktpm2x_status_t last_rc = ST33KTPM2X_ETIMEOUT;
399
400 if (burst_out == NULL)
401 {
402 return ST33KTPM2X_EINVAL;
403 }
404
405 start_ms = HAL_GetTick();
406 do
407 {
408 rc = st33_read_sts(ctx, &sts, &burst);
409 last_rc = rc;
410 last_sts = sts;
411 last_burst = burst;
412 if (rc == ST33KTPM2X_OK)
413 {
414 if ((sts & ST33_STS_VALID) != 0U)
415 {
416 if ((burst > 0U) && (((sts & ST33_STS_DATA_AVAIL) != 0U) || (require_data_avail == 0U)))
417 {
418 *burst_out = burst;
419 return ST33KTPM2X_OK;
420 }
421 }
422 }
423 app_delay_ms(1U);
424 } while ((HAL_GetTick() - start_ms) < ctx->cfg.burst_timeout_ms);
425
426 ST33_TRACE("TPM TIS wait burst timeout data_avail=%u last_rc=%d(%s) sts=0x%02X burst=%u timeout=%lu\r\n",
427 (unsigned int)require_data_avail,
428 (int)last_rc,
429 st33_status_text(last_rc),
430 (unsigned int)last_sts,
431 (unsigned int)last_burst,
432 (unsigned long)ctx->cfg.burst_timeout_ms);
433 return ST33KTPM2X_ETIMEOUT;
434}

491static st33ktpm2x_status_t st33_read_response_stream(st33ktpm2x_t *ctx, uint8_t *dst, uint16_t len)
492{
493 uint16_t got = 0U;
494 uint16_t burst = 0U;
495 uint16_t chunk = 0U;
496 st33ktpm2x_status_t rc;
497
498 while (got < len)
499 {
500 rc = st33_wait_burst(ctx, &burst, 1U);
501 if (rc != ST33KTPM2X_OK)
502 {
503 return rc;
504 }
505
506 chunk = st33_min_u16((uint16_t)(len - got), burst);
507 rc = st33_fifo_read(ctx, &dst[got], chunk);
508 if (rc != ST33KTPM2X_OK)
509 {
510 return rc;
511 }
512 got = (uint16_t)(got + chunk);
513 }
514
515 return ST33KTPM2X_OK;
516}

672st33ktpm2x_status_t st33ktpm2x_request_locality0(st33ktpm2x_t *ctx)
673{
674 uint8_t access = 0U;
675 uint8_t locality = 0U;
676 st33ktpm2x_status_t rc;
677
678 if ((ctx == NULL) || (!ctx->initialized))
679 {
680 return ST33KTPM2X_ESTATE;
681 }
682
683 if (ctx->locality0_acquired)
684 {
685 st33_trace_tis(ctx, "request_locality_cached");
686 return ST33KTPM2X_OK;
687 }
688
689 rc = st33_write_loc_sel(ctx, locality);
690 if (rc != ST33KTPM2X_OK)
691 {
692 return st33_trace_fail(ctx, "write_loc_sel", rc, false);
693 }
694
695 rc = st33_read_access(ctx, &access);
696 if (rc != ST33KTPM2X_OK)
697 {
698 return st33_trace_fail(ctx, "read_access_initial", rc, false);
699 }
700 ST33_TRACE("TPM TIS request locality initial access=0x%02X timeout=%lu\r\n",
701 (unsigned int)access,
702 (unsigned long)ctx->cfg.locality_timeout_ms);
703
704 if ((access & ST33_ACCESS_ACTIVE_LOCALITY) == 0U)
705 {

786st33ktpm2x_status_t st33ktpm2x_transceive(st33ktpm2x_t *ctx,
787 const uint8_t *command,
788 uint16_t command_len,
789 uint8_t *response,
790 uint16_t response_capacity,
791 uint16_t *response_len,
792 uint32_t *tpm_rc)
793{
794 uint8_t sts = 0U;
795 uint16_t burst = 0U;
796 uint16_t sent = 0U;
797 uint16_t chunk = 0U;
798 uint16_t rsp_total = 0U;
799 uint32_t command_code = 0UL;
800 st33ktpm2x_status_t rc;
801
802 if ((ctx == NULL) || (!ctx->initialized) || (command == NULL) || (response == NULL) || (response_len == NULL))
803 {
804 return ST33KTPM2X_EINVAL;
805 }
806 if ((command_len < ST33_TPM_HEADER_SIZE) || (command_len > ST33KTPM2X_CMD_MAX_BYTES))
807 {
808 return ST33KTPM2X_EINVAL;
809 }
810
811 *response_len = 0U;
812 if (tpm_rc != NULL)
813 {
814 *tpm_rc = 0UL;
815 }
816
817 command_code = st33_be32_read(&command[6]);
818 ST33_TRACE("TPM TIS transceive start cc=0x%08lX cmd_len=%u rsp_cap=%u\r\n",
819 (unsigned long)command_code,
820 (unsigned int)command_len,
821 (unsigned int)response_capacity);
822
823 rc = st33ktpm2x_request_locality0(ctx);
824 if (rc != ST33KTPM2X_OK)
825 {
826 return st33_trace_fail(ctx, "request_locality", rc, false);
827 }
828 st33_trace_tis(ctx, "transceive_after_locality");
829
830 rc = st33_write_sts(ctx, ST33_STS_COMMAND_READY);
831 if (rc != ST33KTPM2X_OK)
832 {
833 return st33_trace_fail(ctx, "write_command_ready", rc, true);
834 }
835 st33_trace_tis(ctx, "after_write_command_ready");
836
837 rc = st33_wait_sts_bits(ctx,
838 (uint8_t)(ST33_STS_VALID | ST33_STS_COMMAND_READY),
839 (uint8_t)(ST33_STS_VALID | ST33_STS_COMMAND_READY),
840 ctx->cfg.burst_timeout_ms);
841 if (rc != ST33KTPM2X_OK)
842 {
843 return st33_trace_fail(ctx, "wait_command_ready", rc, true);
844 }
845 st33_trace_tis(ctx, "command_ready");
846
847 while (sent < command_len)
848 {
849 rc = st33_wait_burst(ctx, &burst, 0U);
850 if (rc != ST33KTPM2X_OK)
851 {
852 return st33_trace_fail(ctx, "wait_write_burst", rc, true);
853 }
854
855 chunk = st33_min_u16((uint16_t)(command_len - sent), burst);
856 ST33_TRACE("TPM TIS write FIFO chunk=%u sent_before=%u burst=%u\r\n",
857 (unsigned int)chunk,
858 (unsigned int)sent,
859 (unsigned int)burst);
860 rc = st33_fifo_write(ctx, &command[sent], chunk);
861 if (rc != ST33KTPM2X_OK)
862 {
863 return st33_trace_fail(ctx, "fifo_write", rc, true);
864 }
865
866 sent = (uint16_t)(sent + chunk);
867 rc = st33_read_sts(ctx, &sts, NULL);
868 if (rc != ST33KTPM2X_OK)
869 {
870 return st33_trace_fail(ctx, "read_sts_after_fifo_write", rc, true);
871 }
872 ST33_TRACE("TPM TIS after FIFO sent=%u/%u sts=0x%02X expect=%u\r\n",
873 (unsigned int)sent,
874 (unsigned int)command_len,
875 (unsigned int)sts,
876 ((sts & ST33_STS_EXPECT) != 0U) ? 1U : 0U);
877 if ((sent < command_len) && ((sts & ST33_STS_EXPECT) == 0U))
878 {
879 return st33_trace_fail(ctx, "expect_cleared_before_command_end", ST33KTPM2X_EPROTO, true);
880 }
881 }
882
883 rc = st33_write_sts(ctx, ST33_STS_TPM_GO);
884 if (rc != ST33KTPM2X_OK)
885 {
886 return st33_trace_fail(ctx, "write_tpm_go", rc, true);
887 }
888 st33_trace_tis(ctx, "after_tpm_go");
889
890 rc = st33_wait_sts_bits(ctx,
891 (uint8_t)(ST33_STS_VALID | ST33_STS_DATA_AVAIL),
892 (uint8_t)(ST33_STS_VALID | ST33_STS_DATA_AVAIL),
893 ctx->cfg.burst_timeout_ms);
894 if (rc != ST33KTPM2X_OK)
895 {
896 return st33_trace_fail(ctx, "wait_data_avail", rc, true);
897 }
898 st33_trace_tis(ctx, "data_avail");
899
900 rc = st33_read_response_stream(ctx, response, ST33_TPM_HEADER_SIZE);
901 if (rc != ST33KTPM2X_OK)
902 {
903 return st33_trace_fail(ctx, "read_response_header", rc, true);
904 }
905
906 rsp_total = (uint16_t)st33_be32_read(&response[2]);
907 ST33_TRACE("TPM TIS response header cc=0x%08lX total=%u tpm_rc=0x%08lX\r\n",
908 (unsigned long)command_code,
909 (unsigned int)rsp_total,
910 (unsigned long)st33_be32_read(&response[6]));
911 if ((rsp_total < ST33_TPM_HEADER_SIZE) || (rsp_total > ST33KTPM2X_RSP_MAX_BYTES))
912 {
913 (void)st33_write_sts(ctx, ST33_STS_COMMAND_READY);
914 return st33_trace_fail(ctx, "response_size_invalid", ST33KTPM2X_EPROTO, true);
915 }
916 if (rsp_total > response_capacity)
917 {
918 (void)st33_write_sts(ctx, ST33_STS_COMMAND_READY);
919 return st33_trace_fail(ctx, "response_capacity", ST33KTPM2X_EOVERFLOW, true);
920 }
921
922 if (rsp_total > ST33_TPM_HEADER_SIZE)
923 {
924 rc = st33_read_response_stream(ctx,
925 &response[ST33_TPM_HEADER_SIZE],
926 (uint16_t)(rsp_total - ST33_TPM_HEADER_SIZE));
927 if (rc != ST33KTPM2X_OK)
928 {
929 (void)st33_write_sts(ctx, ST33_STS_COMMAND_READY);
930 return st33_trace_fail(ctx, "read_response_body", rc, true);
931 }
932 }
933
934 *response_len = rsp_total;
935 if (tpm_rc != NULL)
936 {
937 *tpm_rc = st33_be32_read(&response[6]);
938 }
939
940 (void)st33_write_sts(ctx, ST33_STS_COMMAND_READY);
941 (void)st33ktpm2x_release_locality0(ctx);
942
943 if ((tpm_rc != NULL) && (*tpm_rc != ST33KTPM2X_TPM2_RC_SUCCESS))
944 {
945 ST33_TRACE("TPM TIS command TPM_RC cc=0x%08lX tpm_rc=0x%08lX\r\n",
946 (unsigned long)command_code,
947 (unsigned long)*tpm_rc);
948 return ST33KTPM2X_ETPM_RC;
949 }
950
951 ST33_TRACE("TPM TIS transceive OK cc=0x%08lX rsp_len=%u\r\n",
952 (unsigned long)command_code,
953 (unsigned int)*response_len);
954 return ST33KTPM2X_OK;
955}

1096st33ktpm2x_status_t st33ktpm2x_tpm2_pcr_read_sha256(st33ktpm2x_t *ctx,
1097 uint8_t pcr_index,
1098 uint8_t digest_sha256_out[32],
1099 uint32_t *update_counter_out,
1100 uint32_t *tpm_rc)
1101{
1102 uint8_t cmd[20];
1103 uint8_t rsp[ST33KTPM2X_RSP_MAX_BYTES];
1104 uint16_t rsp_len = 0U;
1105 uint16_t off = ST33_TPM_HEADER_SIZE;
1106 uint32_t sel_count;
1107 uint32_t dig_count;
1108 uint8_t sizeof_select;
1109 uint16_t dig_len;
1110 uint32_t i;
1111 st33ktpm2x_status_t rc;
1112
1113 if ((digest_sha256_out == NULL) || (update_counter_out == NULL) || (pcr_index > 23U))
1114 {
1115 return ST33KTPM2X_EINVAL;
1116 }
1117
1118 memset(digest_sha256_out, 0, 32U);
1119 *update_counter_out = 0UL;
1120
1121 st33_be16_write(&cmd[0], ST33KTPM2X_TPM2_ST_NO_SESSIONS);
1122 st33_be32_write(&cmd[2], (uint32_t)sizeof(cmd));
1123 st33_be32_write(&cmd[6], ST33KTPM2X_TPM2_CC_PCR_READ);
1124
1125 st33_be32_write(&cmd[10], 1UL); /* TPML_PCR_SELECTION.count */
1126 st33_be16_write(&cmd[14], ST33KTPM2X_TPM2_ALG_SHA256);
1127 cmd[16] = 3U; /* sizeofSelect */
1128 cmd[17] = 0U;
1129 cmd[18] = 0U;
1130 cmd[19] = 0U;
1131 cmd[17U + (uint8_t)(pcr_index / 8U)] = (uint8_t)(1U << (pcr_index % 8U));
1132
1133 rc = st33ktpm2x_transceive(ctx, cmd, (uint16_t)sizeof(cmd), rsp, (uint16_t)sizeof(rsp), &rsp_len, tpm_rc);
1134 if (rc != ST33KTPM2X_OK)
1135 {
1136 return rc;
1137 }
1138
1139 if (rsp_len < (ST33_TPM_HEADER_SIZE + 4U + 4U + 4U + 2U))
1140 {
1141 return ST33KTPM2X_EPROTO;
1142 }
1143
1144 *update_counter_out = st33_be32_read(&rsp[off]);
1145 off = (uint16_t)(off + 4U);
1146
1147 sel_count = st33_be32_read(&rsp[off]);
1148 off = (uint16_t)(off + 4U);
1149 for (i = 0UL; i < sel_count; i++)
1150 {
1151 if ((uint16_t)(off + 3U) > rsp_len)
1152 {
1153 return ST33KTPM2X_EPROTO;
1154 }
1155 off = (uint16_t)(off + 2U); /* hash */
1156 sizeof_select = rsp[off];
1157 off = (uint16_t)(off + 1U); /* sizeofSelect */
1158 if ((uint16_t)(off + sizeof_select) > rsp_len)
1159 {
1160 return ST33KTPM2X_EPROTO;
1161 }
1162 off = (uint16_t)(off + sizeof_select);
1163 }
1164
1165 if ((uint16_t)(off + 4U + 2U) > rsp_len)
1166 {
1167 return ST33KTPM2X_EPROTO;
1168 }
1169
1170 dig_count = st33_be32_read(&rsp[off]);
1171 off = (uint16_t)(off + 4U);
1172 if (dig_count == 0UL)
1173 {
1174 return ST33KTPM2X_EPROTO;
1175 }
1176
1177 dig_len = st33_be16_read(&rsp[off]);
1178 off = (uint16_t)(off + 2U);
1179 if (dig_len != 32U)
1180 {
1181 return ST33KTPM2X_EPROTO;
1182 }
1183 if ((uint16_t)(off + dig_len) > rsp_len)
1184 {
1185 return ST33KTPM2X_EPROTO;
1186 }
1187
1188 memcpy(digest_sha256_out, &rsp[off], 32U);
1189 return ST33KTPM2X_OK;
1190}
