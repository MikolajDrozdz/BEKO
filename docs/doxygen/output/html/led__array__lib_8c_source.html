54static uint32_t led_array_now_ms(void)
55{
56 osKernelState_t state;
57 uint32_t tick_hz;
58 uint32_t tick_count;
59
60 state = osKernelGetState();
61 if ((state == osKernelRunning) || (state == osKernelLocked))
62 {
63 tick_hz = osKernelGetTickFreq();
64 tick_count = osKernelGetTickCount();
65 if (tick_hz == 0U)
66 {
67 return tick_count;
68 }
69 return (uint32_t)(((uint64_t)tick_count * 1000ULL) / (uint64_t)tick_hz);
70 }
71
72 return HAL_GetTick();
73}

76static uint32_t led_array_get_tim_clock(TIM_TypeDef *tim)
77{
78 uint32_t pclk;
79 uint32_t timer_clock;
80 RCC_ClkInitTypeDef clk_cfg;
81 uint32_t flash_latency;
82
83 HAL_RCC_GetClockConfig(&clk_cfg, &flash_latency);
84
85 if ((tim == TIM2) || (tim == TIM3))
86 {
87 pclk = HAL_RCC_GetPCLK1Freq();
88 if (clk_cfg.APB1CLKDivider == RCC_HCLK_DIV1)
89 {
90 timer_clock = pclk;
91 }
92 else
93 {
94 timer_clock = pclk * 2U;
95 }
96 }
97 else
98 {
99 pclk = HAL_RCC_GetPCLK2Freq();
100 if (clk_cfg.APB2CLKDivider == RCC_HCLK_DIV1)
101 {
102 timer_clock = pclk;
103 }
104 else
105 {
106 timer_clock = pclk * 2U;
107 }
108 }
109
110 return timer_clock;
111}

114static void led_array_timer_base_init(TIM_TypeDef *tim)
115{
116 uint32_t tim_clk;
117 uint32_t ticks_per_period;
118 uint32_t psc;
119
120 tim_clk = led_array_get_tim_clock(tim);
121 ticks_per_period = LED_ARRAY_PWM_FREQ_HZ * LED_ARRAY_PWM_STEPS;
122 if ((ticks_per_period == 0U) || (tim_clk == 0U))
123 {
124 return;
125 }
126
127 psc = tim_clk / ticks_per_period;
128 if (psc == 0U)
129 {
130 psc = 1U;
131 }
132
133 tim->CR1 = 0U;
134 tim->PSC = (uint16_t)(psc - 1U);
135 tim->ARR = (uint32_t)(LED_ARRAY_PWM_STEPS - 1U);
136 tim->EGR = TIM_EGR_UG;
137 tim->CR1 |= TIM_CR1_ARPE;
138}

141static void led_array_timer_pwm_ch_init(TIM_TypeDef *tim, uint8_t channel)
142{
143 if (channel == 1U)
144 {
145 tim->CCMR1 &= ~(TIM_CCMR1_OC1M_Msk | TIM_CCMR1_CC1S_Msk);
146 tim->CCMR1 |= (6U << TIM_CCMR1_OC1M_Pos) | TIM_CCMR1_OC1PE;
147 tim->CCER |= TIM_CCER_CC1E;
148 tim->CCR1 = 0U;
149 }
150 else if (channel == 2U)
151 {
152 tim->CCMR1 &= ~(TIM_CCMR1_OC2M_Msk | TIM_CCMR1_CC2S_Msk);
153 tim->CCMR1 |= (6U << TIM_CCMR1_OC2M_Pos) | TIM_CCMR1_OC2PE;
154 tim->CCER |= TIM_CCER_CC2E;
155 tim->CCR2 = 0U;
156 }
157 else if (channel == 3U)
158 {
159 tim->CCMR2 &= ~(TIM_CCMR2_OC3M_Msk | TIM_CCMR2_CC3S_Msk);
160 tim->CCMR2 |= (6U << TIM_CCMR2_OC3M_Pos) | TIM_CCMR2_OC3PE;
161 tim->CCER |= TIM_CCER_CC3E;
162 tim->CCR3 = 0U;
163 }

220static void led_array_set_hw_duty(uint8_t index, uint8_t pct)
221{
222 uint32_t ccr;
223
224 if (pct > 100U)
225 {
226 pct = 100U;
227 }
228 ccr = ((uint32_t)pct * (LED_ARRAY_PWM_STEPS - 1U)) / 100U;
229
230 switch (index)
231 {
232 case 0U: TIM8->CCR2 = ccr; break;
233 case 1U: TIM1->CCR1 = ccr; break;
234 case 2U: TIM2->CCR3 = ccr; break;
235 case 3U: TIM3->CCR1 = ccr; break;
236 case 4U: TIM3->CCR2 = ccr; break;
237 default: break;
238 }
239}

300static void led_array_effect_fade_update(uint32_t delta_ms)
301{
302 uint8_t i;
303 uint32_t remain;
304 uint32_t level;
305
306 if (s_led.fade.duration_ms == 0U)
307 {
308 s_led.effect = LED_ARRAY_EFFECT_NONE;
309 led_array_effect_none();
310 return;
311 }
312
313 s_led.fade.elapsed_ms += delta_ms;
314 if (s_led.fade.elapsed_ms >= s_led.fade.duration_ms)
315 {
316 for (i = 0U; i < LED_ARRAY_COUNT; i++)
317 {
318 if ((s_led.fade.mask & (uint8_t)(1U << i)) != 0U)
319 {
320 s_led.manual_brightness[i] = 0U;
321 s_led.active_brightness[i] = 0U;
322 }
323 else
324 {
325 s_led.active_brightness[i] = s_led.manual_brightness[i];
326 }
327 }
328
329 s_led.effect = LED_ARRAY_EFFECT_NONE;
330 return;
331 }
332
333 remain = (uint32_t)s_led.fade.duration_ms - s_led.fade.elapsed_ms;
334
335 for (i = 0U; i < LED_ARRAY_COUNT; i++)
336 {
337 if ((s_led.fade.mask & (uint8_t)(1U << i)) != 0U)
338 {
339 level = (100U * remain) / (uint32_t)s_led.fade.duration_ms;
340 s_led.active_brightness[i] = (uint8_t)level;
341 }
342 else
343 {
344 s_led.active_brightness[i] = s_led.manual_brightness[i];
345 }
346 }
347}

350static void led_array_effect_breath_update(uint32_t delta_ms)
351{
352 uint8_t i;
353 uint32_t half;
354 uint32_t phase;
355 uint32_t tri;
356 uint32_t range;
357 uint8_t level;
358
359 if (s_led.breath.period_ms < 2U)
360 {
361 s_led.breath.period_ms = 2U;
362 }
363
364 s_led.breath.elapsed_ms = (s_led.breath.elapsed_ms + delta_ms) % s_led.breath.period_ms;
365 half = s_led.breath.period_ms / 2U;
366 if (half == 0U)
367 {
368 half = 1U;
369 }
370
371 if (s_led.breath.elapsed_ms < half)
372 {
373 phase = s_led.breath.elapsed_ms;
374 }
375 else
376 {
377 phase = (uint32_t)s_led.breath.period_ms - s_led.breath.elapsed_ms;
378 }
379
380 tri = (phase * 100U) / half;
381 if (tri > 100U)
382 {
383 tri = 100U;
384 }
385
386 range = (uint32_t)s_led.breath.max_pct - s_led.breath.min_pct;
387 level = (uint8_t)(s_led.breath.min_pct + ((range * tri) / 100U));
388
389 for (i = 0U; i < LED_ARRAY_COUNT; i++)
390 {
391 if ((s_led.breath.mask & (uint8_t)(1U << i)) != 0U)
392 {
393 s_led.active_brightness[i] = level;
394 }
395 else
396 {
397 s_led.active_brightness[i] = s_led.manual_brightness[i];
398 }
399 }

403static void led_array_effect_rainbow_update(uint32_t delta_ms)
404{
405 uint8_t i;
406 uint8_t phase;
407 uint8_t tri;
408 uint32_t range;
409 uint8_t level;
410 const uint8_t phase_step = (uint8_t)(256U / LED_ARRAY_COUNT);
411
412 if (s_led.rainbow.step_ms == 0U)
413 {
414 s_led.rainbow.step_ms = 1U;
415 }
416
417 s_led.rainbow.phase_accum_ms = (uint16_t)(s_led.rainbow.phase_accum_ms + delta_ms);
418 while (s_led.rainbow.phase_accum_ms >= s_led.rainbow.step_ms)
419 {
420 s_led.rainbow.phase_accum_ms = (uint16_t)(s_led.rainbow.phase_accum_ms - s_led.rainbow.step_ms);
421 s_led.rainbow.phase_base = (uint8_t)(s_led.rainbow.phase_base + LED_ARRAY_RAINBOW_PHASE_INC);
422 }
423
424 range = (uint32_t)s_led.rainbow.max_pct - s_led.rainbow.min_pct;
425 for (i = 0U; i < LED_ARRAY_COUNT; i++)
426 {
427 phase = (uint8_t)(s_led.rainbow.phase_base + (uint8_t)(phase_step * i));
428 tri = led_array_tri8(phase);
429 level = (uint8_t)(s_led.rainbow.min_pct + ((range * tri) / 254U));
430 s_led.active_brightness[i] = level;
431 }
432}

435static void led_array_tick_core(uint32_t delta_ms)
436{
437 switch (s_led.effect)
438 {
439 case LED_ARRAY_EFFECT_FADE_OUT:
440 led_array_effect_fade_update(delta_ms);
441 break;
442 case LED_ARRAY_EFFECT_BREATH:
443 led_array_effect_breath_update(delta_ms);
444 break;
445 case LED_ARRAY_EFFECT_RAINBOW:
446 led_array_effect_rainbow_update(delta_ms);
447 break;
448 case LED_ARRAY_EFFECT_NONE:
449 default:
450 led_array_effect_none();
451 break;
452 }
453
454 led_array_apply_outputs();
455}

527void led_array_process(void)
528{
529 uint32_t now;
530
531 if (!s_led.initialized || !s_led.timer_running)
532 {
533 return;
534 }
535
536 now = led_array_now_ms();
537 while ((now - s_led.last_tick_ms) >= s_led.tick_period_ms)
538 {
539 s_led.last_tick_ms += s_led.tick_period_ms;
540 led_array_tick_core(s_led.tick_period_ms);
541 }
