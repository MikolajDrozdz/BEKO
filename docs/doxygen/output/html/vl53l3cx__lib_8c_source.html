87static bool tof_low_level_init(void)
88{
89 VL53LX_Error status;
90
91 status = VL53LX_WaitDeviceBooted(&s_tof_obj);
92 if (status != VL53LX_ERROR_NONE)
93 {
94 printf("ToF init: WaitDeviceBooted failed (%ld)\r\n", (long)status);
95 return false;
96 }
97
98 status = VL53LX_DataInit(&s_tof_obj);
99 if (status != VL53LX_ERROR_NONE)
100 {
101 printf("ToF init: DataInit failed (%ld)\r\n", (long)status);
102 return false;
103 }
104
105 status = VL53LX_PerformRefSpadManagement(&s_tof_obj);
106 if ((status != VL53LX_ERROR_NONE) && !tof_is_non_fatal_ref_spad_status(status))
107 {
108 printf("ToF init: RefSpadManagement failed (%ld)\r\n", (long)status);
109 return false;
110 }
111
112 if (tof_is_non_fatal_ref_spad_status(status))
113 {
114 printf("ToF init: RefSpadManagement warning (%ld), continuing\r\n", (long)status);
115 }
116
117 s_tof_obj.IsRanging = 0U;
118 s_tof_obj.IsBlocking = 0U;
119 s_tof_obj.IsContinuous = 0U;
120 s_tof_obj.IsAmbientEnabled = 0U;
121 s_tof_obj.IsSignalEnabled = 0U;
122 s_tof_obj.IsInitialized = 1U;
123
124 return true;
125}

127bool tof_init(void)
128{
129 VL53L3CX_IO_t io_ctx;
130 VL53L3CX_ProfileConfig_t profile;
131 uint32_t sensor_id;
132
133 if (s_tof_initialized == 1U)
134 {
135 return true;
136 }
137
138 s_tof_started = 0U;
139 s_first_reading = true;
140 memset(&s_tof_obj, 0, sizeof(s_tof_obj));
141
142 (void)tof_reset_and_boot();
143
144 io_ctx.Init = tof_bus_init;
145 io_ctx.DeInit = tof_bus_deinit;
146 io_ctx.Address = TOF_I2C_ADDRESS_8BIT;
147 io_ctx.WriteReg = tof_bus_write;
148 io_ctx.ReadReg = tof_bus_read;
149 io_ctx.GetTick = tof_bus_get_tick;
150
151 if (VL53L3CX_RegisterBusIO(&s_tof_obj, &io_ctx) != VL53L3CX_OK)
152 {
153 return false;
154 }
155
156 sensor_id = 0U;
157 if (VL53L3CX_ReadID(&s_tof_obj, &sensor_id) != VL53L3CX_OK)
158 {
159 printf("ToF init: ReadID failed\r\n");
160 return false;
161 }
162
163 if (sensor_id != VL53L3CX_ID)
164 {
165 printf("ToF init: unexpected sensor ID 0x%04lX\r\n", (unsigned long)sensor_id);
166 return false;
167 }
168
169 if (!tof_low_level_init())
170 {
171 return false;
172 }
173
174 profile.RangingProfile = VL53L3CX_PROFILE_LONG;
175 profile.TimingBudget = TOF_TIMING_BUDGET_MS;
176 profile.Frequency = 0U;
177 profile.EnableAmbient = 0U;
178 profile.EnableSignal = 0U;
179
180 if (VL53L3CX_ConfigProfile(&s_tof_obj, &profile) != VL53L3CX_OK)
181 {
182 return false;
183 }
184
185 s_tof_initialized = 1U;
186 return true;
187}

190static bool tof_activate(void)
191{
192 if (s_tof_started == 1U)
193 {
194 return true;
195 }
196
197 /*
198 * Blocking mode prevents timeout/stale reads when the caller polls faster
199 * than the sensor produces new samples.
200 */
201 if (VL53L3CX_Start(&s_tof_obj, VL53L3CX_MODE_BLOCKING_CONTINUOUS) != VL53L3CX_OK)
202 {
203 return false;
204 }
205
206 s_tof_started = 1U;
207 return true;
208}

210int32_t tof_get_distance(void)
211{
212 int32_t status;
213 uint32_t target_count;
214
215 if (!tof_init())
216 {
217 return -1;
218 }
219
220 if (!tof_activate())
221 {
222 return -1;
223 }
224
225 memset(&result, 0, sizeof(result));
226
227 if (s_first_reading)
228 {
229 VL53L3CX_GetDistance(&s_tof_obj, &result);
230 s_first_reading = false;
231 app_delay_ms(100U);
232 }
233
234 status = VL53L3CX_GetDistance(&s_tof_obj, &result);
235 if (status != VL53L3CX_OK)
236 {
237 return -1;
238 }
239
240 target_count = result.ZoneResult[0].NumberOfTargets;
241
242 if ((result.ZoneResult[0].Status[0] == 0U) &&
243 (result.ZoneResult[0].Distance[0] > 0U) &&
244 target_count > 0)
245 {
246 return (int32_t)result.ZoneResult[0].Distance[0];
247 }
248 else
249 {
250 return -1;
251 }
252 return -1;
253}
