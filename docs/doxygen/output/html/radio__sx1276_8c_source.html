11{

13 {
14 return false;

16

27bool sx1276_read_reg(const sx1276_bus_t *bus, uint8_t reg, uint8_t *value)
28{
29 uint8_t tx_buf[2];
30 uint8_t rx_buf[2];
31
32 if ((value == NULL) || !sx1276_select(bus))
33 {
34 return false;
35 }
36
37 tx_buf[0] = (uint8_t)(reg & SX1276_SPI_READ_MASK);
38 tx_buf[1] = 0x00U;
39
40 if (HAL_SPI_TransmitReceive(bus->hspi, tx_buf, rx_buf, 2U, bus->spi_timeout_ms) != HAL_OK)
41 {
42 sx1276_deselect(bus);
43 return false;
44 }
45
46 sx1276_deselect(bus);
47 *value = rx_buf[1];
48 return true;
49}

135bool sx1276_set_frequency(const sx1276_bus_t *bus, uint32_t frequency_hz)
136{
137 uint64_t frf;
138 uint8_t frf_msb;
139 uint8_t frf_mid;
140 uint8_t frf_lsb;
141
142 frf = (((uint64_t)frequency_hz) << 19) / 32000000ULL;
143 frf_msb = (uint8_t)((frf >> 16) & 0xFFU);
144 frf_mid = (uint8_t)((frf >> 8) & 0xFFU);
145 frf_lsb = (uint8_t)(frf & 0xFFU);
146
147 return sx1276_write_reg(bus, SX1276_REG_FRF_MSB, frf_msb) &&
148 sx1276_write_reg(bus, SX1276_REG_FRF_MID, frf_mid) &&
149 sx1276_write_reg(bus, SX1276_REG_FRF_LSB, frf_lsb);
150}

152bool sx1276_set_pa_output_power(const sx1276_bus_t *bus, int8_t tx_power_dbm)
153{
154 uint8_t pa_config;
155 uint8_t pa_dac;
156 int8_t out_power;
157
158 if (tx_power_dbm < 2)
159 {
160 tx_power_dbm = 2;
161 }
162 else if (tx_power_dbm > 20)
163 {
164 tx_power_dbm = 20;
165 }
166
167 if (tx_power_dbm > 17)
168 {
169 pa_dac = SX1276_PA_DAC_ENABLE_20DBM;
170 out_power = (int8_t)(tx_power_dbm - 5);
171 }
172 else
173 {
174 pa_dac = SX1276_PA_DAC_DISABLE_17DBM;
175 out_power = (int8_t)(tx_power_dbm - 2);
176 }
177
178 pa_config = (uint8_t)(0x80U | ((uint8_t)out_power & 0x0FU));
179
180 return sx1276_write_reg(bus, SX1276_REG_PA_DAC, pa_dac) &&
181 sx1276_write_reg(bus, SX1276_REG_PA_CONFIG, pa_config);
182}

184bool sx1276_set_lora_modem(const sx1276_bus_t *bus,
185 uint8_t bandwidth_code,
186 uint8_t spreading_factor,
187 uint8_t coding_rate_denominator,
188 bool implicit_header,
189 bool crc_on,
190 bool low_datarate_optimize)
191{
192 uint8_t modem_config_1;
193 uint8_t modem_config_2;
194 uint8_t modem_config_3;
195 uint8_t coding_rate_field;
196 uint8_t detect_optimize;
197
198 if ((bandwidth_code > 9U) ||
199 (spreading_factor < 6U) || (spreading_factor > 12U) ||
200 (coding_rate_denominator < 5U) || (coding_rate_denominator > 8U))
201 {
202 return false;
203 }
204
205 coding_rate_field = (uint8_t)(coding_rate_denominator - 4U);
206
207 modem_config_1 = (uint8_t)((bandwidth_code << 4) |
208 (coding_rate_field << 1) |
209 (implicit_header ? 1U : 0U));
210
211 modem_config_2 = (uint8_t)((spreading_factor << 4) |
212 (crc_on ? 0x04U : 0x00U));
213
214 modem_config_3 = (uint8_t)((low_datarate_optimize ? 0x08U : 0x00U) | 0x04U);
215
216 if (!sx1276_write_reg(bus, SX1276_REG_MODEM_CONFIG_1, modem_config_1) ||
217 !sx1276_write_reg(bus, SX1276_REG_MODEM_CONFIG_2, modem_config_2) ||
218 !sx1276_write_reg(bus, SX1276_REG_MODEM_CONFIG_3, modem_config_3))
219 {
220 return false;
221 }
222
223 if (!sx1276_read_reg(bus, SX1276_REG_DETECTION_OPTIMIZE, &detect_optimize))
224 {
225 return false;
226 }
227
228 detect_optimize = (uint8_t)(detect_optimize & 0xF8U);
229 if (spreading_factor == 6U)
230 {
231 detect_optimize |= 0x05U;
232 return sx1276_write_reg(bus, SX1276_REG_DETECTION_OPTIMIZE, detect_optimize) &&
233 sx1276_write_reg(bus, SX1276_REG_DETECTION_THRESHOLD, 0x0CU);
234 }
235
236 detect_optimize |= 0x03U;
237 return sx1276_write_reg(bus, SX1276_REG_DETECTION_OPTIMIZE, detect_optimize) &&
238 sx1276_write_reg(bus, SX1276_REG_DETECTION_THRESHOLD, 0x0AU);
239}
