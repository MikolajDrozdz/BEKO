79static bool radio_cfg_valid(const radio_lora_cfg_t *cfg)
80{
81 if ((cfg == NULL) ||
82 (cfg->frequency_hz < 137000000UL) || (cfg->frequency_hz > 1020000000UL) ||
83 ((uint8_t)cfg->bandwidth > (uint8_t)RADIO_LORA_BW_500_KHZ) ||
84 (cfg->spreading_factor < 6U) || (cfg->spreading_factor > 12U) ||
85 (cfg->coding_rate < 5U) || (cfg->coding_rate > 8U) ||
86 (cfg->preamble_len == 0U))
87 {
88 return false;
89 }
90
91 if (cfg->implicit_header && (cfg->payload_len == 0U))
92 {
93 return false;
94 }
95
96 if ((cfg->spreading_factor == 6U) && !cfg->implicit_header)
97 {
98 return false;
99 }
100
101 return true;
102}

127static bool radio_apply_lora_config(const radio_lora_cfg_t *cfg)

146 sx1276_set_fifo_base_addrs(&s_radio.bus,
147 RADIO_SX1276_FIFO_TX_BASE_ADDR,

149 sx1276_set_payload_length(&s_radio.bus, cfg->payload_len) &&
150 sx1276_write_reg(&s_radio.bus, SX1276_REG_LNA, 0x23U) &&
151 sx1276_write_reg(&s_radio.bus, SX1276_REG_IRQ_FLAGS_MASK, 0x00U);

179static bool radio_wait_for_tx_enter_or_done(uint8_t *op_mode_snapshot)
180{
181 uint8_t op_mode;
182 uint8_t irq_flags;
183 uint8_t retry;
184
185 for (retry = 0U; retry < 5U; retry++)
186 {
187 if (sx1276_read_reg(&s_radio.bus, SX1276_REG_OP_MODE, &op_mode))
188 {
189 if (op_mode_snapshot != NULL)
190 {
191 *op_mode_snapshot = op_mode;
192 }
193
194 op_mode &= SX1276_OPMODE_MODE_MASK;
195 if ((op_mode == SX1276_MODE_TX) || (op_mode == SX1276_MODE_FSTX))
196 {
197 return true;
198 }
199 }
200
201 if (sx1276_get_irq_flags(&s_radio.bus, &irq_flags))
202 {
203 if ((irq_flags & SX1276_IRQ_TX_DONE) != 0U)
204 {
205 return true;
206 }
207 }
208
209 app_delay_ms(1U);
210 }
211
212 return false;
213}

216static void radio_handle_exti_pin(uint16_t pin)
217{
218 uint8_t i;
219
220 if (!s_radio.initialized)
221 {
222 return;
223 }
224
225 for (i = 0U; i < 6U; i++)
226 {
227 if ((i == 2U) && (RADIO_LIB_ENABLE_DIO2_EXTI == 0))
228 {
229 continue;
230 }
231 if ((i == 3U) && (RADIO_LIB_ENABLE_DIO3_EXTI == 0))
232 {
233 continue;
234 }
235 if ((i == 4U) && (RADIO_LIB_ENABLE_DIO4_EXTI == 0))
236 {
237 continue;
238 }
239 if ((i == 5U) && (RADIO_LIB_ENABLE_DIO5_EXTI == 0))
240 {
241 continue;
242 }
243
244 if (s_radio.hw.dio[i].pin == pin)
245 {
246 uint32_t key = radio_irq_save();
247 s_radio.dio_pending_mask |= (uint8_t)(1U << i);
248 radio_irq_restore(key);
249 break;
250 }
251 }

255static radio_status_t radio_read_rx_packet(void)
256{
257 uint8_t current_addr;
258 uint8_t rx_len;
259 int8_t snr_raw;
260 uint8_t rssi_raw;
261 bool rf_ok;
262
263 rf_ok = sx1276_get_rx_current_addr(&s_radio.bus, &current_addr) &&
264 sx1276_get_rx_nb_bytes(&s_radio.bus, &rx_len) &&
265 sx1276_set_fifo_addr_ptr(&s_radio.bus, current_addr);
266 if (!rf_ok)
267 {
268 return RADIO_EHW;
269 }
270
271 if (rx_len > RADIO_LIB_MAX_PAYLOAD)
272 {
273 s_radio.last_packet.valid = false;
274 return RADIO_EBUS;
275 }
276
277 if (!sx1276_read_fifo(&s_radio.bus, s_radio.last_packet.data, rx_len))
278 {
279 return RADIO_EHW;
280 }
281
282 if (!sx1276_get_packet_snr_raw(&s_radio.bus, &snr_raw) ||
283 !sx1276_get_packet_rssi_raw(&s_radio.bus, &rssi_raw))
284 {
285 return RADIO_EHW;
286 }
287
288 s_radio.last_packet.length = rx_len;
289 s_radio.last_packet.snr_db = (int8_t)(snr_raw / 4);
290 s_radio.last_packet.rssi_dbm = (int16_t)((int16_t)rssi_raw - 157);
291 s_radio.last_packet.timestamp_ms = HAL_GetTick();
292 s_radio.last_packet.valid = true;
293
294 return RADIO_OK;
295}

368radio_status_t radio_lora_init(const radio_hw_cfg_t *hw,
369 const radio_lora_cfg_t *cfg,
370 radio_event_cb_t cb,
371 void *user_ctx)
372{
373 uint8_t version;
374
375 if ((hw == NULL) || (hw->hspi == NULL) ||
376 (hw->nss.port == NULL) || (hw->reset.port == NULL) ||
377 !radio_cfg_valid(cfg))
378 {
379 return RADIO_EINVAL;
380 }
381
382 memset(&s_radio, 0, sizeof(s_radio));
383
384 s_radio.hw = *hw;
385 s_radio.lora = *cfg;
386 s_radio.callback = cb;
387 s_radio.callback_ctx = user_ctx;
388 s_radio.bus.hspi = hw->hspi;
389 s_radio.bus.nss_port = hw->nss.port;
390 s_radio.bus.nss_pin = hw->nss.pin;
391 s_radio.bus.spi_timeout_ms = hw->spi_timeout_ms;
392 s_radio.rx_single_timeout_symbols = RADIO_DEFAULT_RX_TIMEOUT_SYMBOLS;
393 s_radio.state = RADIO_STATE_UNINIT;
394
395 radio_hw_reset();
396
397 if (!sx1276_set_op_mode(&s_radio.bus, (uint8_t)(SX1276_OPMODE_LONG_RANGE_MODE | SX1276_MODE_SLEEP)))
398 {
399 return RADIO_EHW;
400 }
401
402 app_delay_ms(1U);
403
404 if (!sx1276_get_version(&s_radio.bus, &version) || (version != SX1276_VERSION_ID))
405 {
406 return RADIO_EHW;
407 }
408
409 if (!radio_apply_lora_config(&s_radio.lora) ||
410 !sx1276_set_fifo_addr_ptr(&s_radio.bus, RADIO_SX1276_FIFO_TX_BASE_ADDR) ||
411 !sx1276_clear_irq_flags(&s_radio.bus, 0xFFU) ||
412 !sx1276_set_op_mode(&s_radio.bus, (uint8_t)(SX1276_OPMODE_LONG_RANGE_MODE | SX1276_MODE_STDBY)))
413 {
414 return RADIO_EHW;
415 }
416
417#if RADIO_LIB_ENABLE_DIO5_EXTI == 0
418 __HAL_GPIO_EXTI_CLEAR_IT(RFM_DIO5_EXIT_8_Pin);
419 HAL_NVIC_DisableIRQ(RFM_DIO5_EXIT_8_EXTI_IRQn);
420#endif
421#if RADIO_LIB_ENABLE_DIO4_EXTI == 0
422 __HAL_GPIO_EXTI_CLEAR_IT(RFM_DIO4_EXIT_13_Pin);
423 HAL_NVIC_DisableIRQ(RFM_DIO4_EXIT_13_EXTI_IRQn);
424#endif
425#if RADIO_LIB_ENABLE_DIO3_EXTI == 0
426 __HAL_GPIO_EXTI_CLEAR_IT(RFM_DIO3_EXIT_14_Pin);
427 HAL_NVIC_DisableIRQ(RFM_DIO3_EXIT_14_EXTI_IRQn);
428#endif
429#if RADIO_LIB_ENABLE_DIO2_EXTI == 0
430 __HAL_GPIO_EXTI_CLEAR_IT(RFM_DIO2_EXIT_15_Pin);
431 HAL_NVIC_DisableIRQ(RFM_DIO2_EXIT_15_EXTI_IRQn);
432#endif
433
434 s_radio.initialized = true;
435 s_radio.state = RADIO_STATE_STANDBY;
436 return RADIO_OK;
437}

580void radio_lora_process(void)
581{
582 uint8_t pending;
583 uint8_t irq_flags;
584 uint8_t op_mode;
585 uint32_t events = RADIO_EVENT_NONE;
586 uint32_t key;
587
588 if (!s_radio.initialized)
589 {
590 return;
591 }
592
593 key = radio_irq_save();
594 pending = s_radio.dio_pending_mask;
595 s_radio.dio_pending_mask = 0U;
596 radio_irq_restore(key);
597
598 /*
599 * Główna ścieżka: zdarzenia z EXTI (pending != 0).
600 * Fallback: podczas TX sprawdzaj flagi IRQ pollingiem, jeśli zbocze DIO0
601 * zostało pominięte.
602 */
603 if ((pending == 0U) &&
604 (s_radio.hw.dio[0].port != NULL) &&
605 (HAL_GPIO_ReadPin(s_radio.hw.dio[0].port, s_radio.hw.dio[0].pin) == GPIO_PIN_SET))
606 {
607 pending |= 0x01U;
608 }
609
610 if ((pending == 0U) && (s_radio.state != RADIO_STATE_TX))
611 {
612 return;
613 }
614
615 if (!radio_read_irq_flags_retry(&irq_flags))
616 {
617 events |= RADIO_EVENT_HW_ERROR;
618 goto done;
619 }
620
621 if (irq_flags == 0U)
622 {
623 /*
624 * Ścieżka recovery: jeśli zbocze TX_DONE zostało pominięte, układ może
625 * być już poza TX, mimo że stan software nadal wskazuje TX.
626 */
627 if ((s_radio.state == RADIO_STATE_TX) &&
628 sx1276_read_reg(&s_radio.bus, SX1276_REG_OP_MODE, &op_mode))
629 {
630 op_mode &= SX1276_OPMODE_MODE_MASK;
631 if ((op_mode != SX1276_MODE_TX) && (op_mode != SX1276_MODE_FSTX))
632 {
633 events |= RADIO_EVENT_TX_DONE;
634 radio_resume_after_tx();
635 }
636 }
637 goto done;
638 }
639
640 if (!sx1276_clear_irq_flags(&s_radio.bus, irq_flags))
641 {
642 events |= RADIO_EVENT_HW_ERROR;
643 }
644
645 if ((irq_flags & SX1276_IRQ_PAYLOAD_CRC_ERROR) != 0U)
646 {
647 events |= RADIO_EVENT_CRC_ERR;
648 }
649
650 if ((irq_flags & SX1276_IRQ_RX_TIMEOUT) != 0U)
651 {
652 events |= RADIO_EVENT_RX_TIMEOUT;
653 if (s_radio.state == RADIO_STATE_RX_SINGLE)
654 {
655 (void)radio_lora_standby();
656 }
657 }
658
659 if ((irq_flags & SX1276_IRQ_RX_DONE) != 0U)
660 {
661 if ((irq_flags & SX1276_IRQ_PAYLOAD_CRC_ERROR) == 0U)
662 {
663 radio_status_t read_status = radio_read_rx_packet();
664 if (read_status == RADIO_OK)
665 {
666 events |= RADIO_EVENT_RX_DONE;
667 }
668 else if (read_status == RADIO_EBUS)
669 {
670 events |= RADIO_EVENT_FIFO_OVERRUN;
671 }
672 else
673 {
674 events |= RADIO_EVENT_HW_ERROR;
675 }
676 }
677
678 if (s_radio.state == RADIO_STATE_RX_SINGLE)
679 {
680 (void)radio_lora_standby();
681 }
682 }
683
684 if ((irq_flags & SX1276_IRQ_TX_DONE) != 0U)
685 {
686 events |= RADIO_EVENT_TX_DONE;
687 radio_resume_after_tx();
688 }
689
690 if ((irq_flags & SX1276_IRQ_CAD_DONE) != 0U)
691 {
692 events |= RADIO_EVENT_CAD_DONE;
693 }
694
695 if ((irq_flags & SX1276_IRQ_CAD_DETECTED) != 0U)
696 {
697 events |= RADIO_EVENT_CAD_DETECTED;
698 }
699
700done:
701 if (events != RADIO_EVENT_NONE)
702 {
703 radio_set_event_flags(events);
704 if (s_radio.callback != NULL)
705 {
706 s_radio.callback(events, s_radio.callback_ctx);
707 }
708 }

717 events = s_radio.event_flags;
718 s_radio.event_flags = 0U;
719 radio_irq_restore(key);
720
721 return events;
722}
