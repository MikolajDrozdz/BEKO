207static bool laviet_crypto_log_mismatch(const char *stage,
208 const uint8_t *expected,
209 const uint8_t *actual,
210 uint16_t len)
211{
212 uint16_t i;
213
214 if ((expected == NULL) || (actual == NULL))
215 {
216 printf("CRYPTO: self-test failed stage=%s reason=NULL_compare_buffer\r\n",
217 (stage != NULL) ? stage : "?");
218 return false;
219 }
220
221 for (i = 0U; i < len; i++)
222 {
223 if (expected[i] != actual[i])
224 {
225 printf("CRYPTO: self-test mismatch stage=%s index=%u expected=0x%02X actual=0x%02X len=%u\r\n",
226 (stage != NULL) ? stage : "?",
227 (unsigned int)i,
228 (unsigned int)expected[i],
229 (unsigned int)actual[i],
230 (unsigned int)len);
231 return false;
232 }
233 }
234
235 return true;
236}

279static void laviet_sha256_transform(laviet_sha256_ctx_t *ctx, const uint8_t data[64])
280{
281 uint32_t m[64];
282 uint32_t a;
283 uint32_t b;
284 uint32_t c;
285 uint32_t d;
286 uint32_t e;
287 uint32_t f;
288 uint32_t g;
289 uint32_t h;
290 uint32_t i;
291
292 for (i = 0U; i < 16U; i++)
293 {
294 m[i] = ((uint32_t)data[i * 4U] << 24) |
295 ((uint32_t)data[(i * 4U) + 1U] << 16) |
296 ((uint32_t)data[(i * 4U) + 2U] << 8) |
297 (uint32_t)data[(i * 4U) + 3U];
298 }
299 for (i = 16U; i < 64U; i++)
300 {
301 uint32_t s0 = laviet_rotr32(m[i - 15U], 7U) ^
302 laviet_rotr32(m[i - 15U], 18U) ^
303 (m[i - 15U] >> 3);
304 uint32_t s1 = laviet_rotr32(m[i - 2U], 17U) ^
305 laviet_rotr32(m[i - 2U], 19U) ^
306 (m[i - 2U] >> 10);
307 m[i] = m[i - 16U] + s0 + m[i - 7U] + s1;
308 }
309
310 a = ctx->state[0];
311 b = ctx->state[1];
312 c = ctx->state[2];
313 d = ctx->state[3];
314 e = ctx->state[4];
315 f = ctx->state[5];
316 g = ctx->state[6];
317 h = ctx->state[7];
318
319 for (i = 0U; i < 64U; i++)
320 {
321 uint32_t s1 = laviet_rotr32(e, 6U) ^ laviet_rotr32(e, 11U) ^ laviet_rotr32(e, 25U);
322 uint32_t ch = (e & f) ^ ((~e) & g);
323 uint32_t temp1 = h + s1 + ch + s_sha256_k[i] + m[i];
324 uint32_t s0 = laviet_rotr32(a, 2U) ^ laviet_rotr32(a, 13U) ^ laviet_rotr32(a, 22U);
325 uint32_t maj = (a & b) ^ (a & c) ^ (b & c);
326 uint32_t temp2 = s0 + maj;
327
328 h = g;
329 g = f;
330 f = e;
331 e = d + temp1;
332 d = c;
333 c = b;
334 b = a;
335 a = temp1 + temp2;
336 }
337
338 ctx->state[0] += a;
339 ctx->state[1] += b;
340 ctx->state[2] += c;
341 ctx->state[3] += d;
342 ctx->state[4] += e;
343 ctx->state[5] += f;
344 ctx->state[6] += g;
345 ctx->state[7] += h;
346}

364static void laviet_sha256_update(laviet_sha256_ctx_t *ctx, const uint8_t *data, uint16_t len)
365{
366 uint16_t i;
367
368 if ((ctx == NULL) || ((data == NULL) && (len > 0U)))
369 {
370 return;
371 }
372
373 for (i = 0U; i < len; i++)
374 {
375 ctx->data[ctx->datalen] = data[i];
376 ctx->datalen++;
377 if (ctx->datalen == 64U)
378 {
379 laviet_sha256_transform(ctx, ctx->data);
380 ctx->bitlen += 512ULL;
381 ctx->datalen = 0U;
382 }
383 }
384}

387static void laviet_sha256_final(laviet_sha256_ctx_t *ctx, uint8_t hash[LAVIET_SHA256_LEN])
388{
389 uint32_t i = ctx->datalen;
390 uint8_t j;
391
392 if (ctx->datalen < 56U)
393 {
394 ctx->data[i++] = 0x80U;
395 while (i < 56U)
396 {
397 ctx->data[i++] = 0x00U;
398 }
399 }
400 else
401 {
402 ctx->data[i++] = 0x80U;
403 while (i < 64U)
404 {
405 ctx->data[i++] = 0x00U;
406 }
407 laviet_sha256_transform(ctx, ctx->data);
408 memset(ctx->data, 0, 56U);
409 }
410
411 ctx->bitlen += ((uint64_t)ctx->datalen * 8ULL);
412 ctx->data[63] = (uint8_t)ctx->bitlen;
413 ctx->data[62] = (uint8_t)(ctx->bitlen >> 8);
414 ctx->data[61] = (uint8_t)(ctx->bitlen >> 16);
415 ctx->data[60] = (uint8_t)(ctx->bitlen >> 24);
416 ctx->data[59] = (uint8_t)(ctx->bitlen >> 32);
417 ctx->data[58] = (uint8_t)(ctx->bitlen >> 40);
418 ctx->data[57] = (uint8_t)(ctx->bitlen >> 48);
419 ctx->data[56] = (uint8_t)(ctx->bitlen >> 56);
420 laviet_sha256_transform(ctx, ctx->data);
421
422 for (i = 0U; i < 8U; i++)
423 {
424 for (j = 0U; j < 4U; j++)
425 {
426 hash[(i * 4U) + j] = (uint8_t)(ctx->state[i] >> (24U - (j * 8U)));
427 }
428 }
429}

438static void laviet_aes_key_expansion(const uint8_t key[16], uint8_t round_key[176])
439{
440 uint8_t i;
441 uint8_t j;
442 uint8_t temp[4];
443
444 memcpy(round_key, key, 16U);
445 for (i = 4U; i < 44U; i++)
446 {
447 for (j = 0U; j < 4U; j++)
448 {
449 temp[j] = round_key[((i - 1U) * 4U) + j];
450 }
451
452 if ((i % 4U) == 0U)
453 {
454 uint8_t t = temp[0];
455 temp[0] = (uint8_t)(s_aes_sbox[temp[1]] ^ s_aes_rcon[i / 4U]);
456 temp[1] = s_aes_sbox[temp[2]];
457 temp[2] = s_aes_sbox[temp[3]];
458 temp[3] = s_aes_sbox[t];
459 }
460
461 for (j = 0U; j < 4U; j++)
462 {
463 round_key[(i * 4U) + j] = (uint8_t)(round_key[((i - 4U) * 4U) + j] ^ temp[j]);
464 }
465 }
466}

516static void laviet_aes_mix_columns(uint8_t state[16])
517{
518 uint8_t i;
519
520 for (i = 0U; i < 4U; i++)
521 {
522 uint8_t *col = &state[i * 4U];
523 uint8_t t = (uint8_t)(col[0] ^ col[1] ^ col[2] ^ col[3]);
524 uint8_t tmp = col[0];
525 uint8_t tm;
526
527 tm = (uint8_t)(col[0] ^ col[1]);
528 tm = laviet_aes_xtime(tm);
529 col[0] ^= (uint8_t)(tm ^ t);
530 tm = (uint8_t)(col[1] ^ col[2]);
531 tm = laviet_aes_xtime(tm);
532 col[1] ^= (uint8_t)(tm ^ t);
533 tm = (uint8_t)(col[2] ^ col[3]);
534 tm = laviet_aes_xtime(tm);
535 col[2] ^= (uint8_t)(tm ^ t);
536 tm = (uint8_t)(col[3] ^ tmp);
537 tm = laviet_aes_xtime(tm);
538 col[3] ^= (uint8_t)(tm ^ t);
539 }
540}

543static void laviet_aes_encrypt_block(const uint8_t in[16],
544 uint8_t out[16],
545 const uint8_t round_key[176])
546{
547 uint8_t state[16];
548 uint8_t round;
549
550 memcpy(state, in, 16U);
551 laviet_aes_add_round_key(state, round_key);
552
553 for (round = 1U; round < 10U; round++)
554 {
555 laviet_aes_sub_bytes(state);
556 laviet_aes_shift_rows(state);
557 laviet_aes_mix_columns(state);
558 laviet_aes_add_round_key(state, &round_key[round * 16U]);
559 }
560
561 laviet_aes_sub_bytes(state);
562 laviet_aes_shift_rows(state);
563 laviet_aes_add_round_key(state, &round_key[160]);
564 memcpy(out, state, 16U);
565 laviet_secure_zero(state, sizeof(state));
566}

569static void laviet_make_counter_block(const laviet_frame_t *frame,
570 uint16_t block_index,
571 uint8_t out[16])
572{
573 out[0] = (uint8_t)'L';
574 out[1] = (uint8_t)'V';
575 out[2] = (uint8_t)'1';
576 out[3] = 0U;
577 out[4] = (uint8_t)(frame->src_id >> 8);
578 out[5] = (uint8_t)frame->src_id;
579 out[6] = (uint8_t)(frame->dst_id >> 8);
