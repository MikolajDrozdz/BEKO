72bool app_i2c_lock(uint32_t timeout_ms)
73{
74 uint32_t timeout;
75 osKernelState_t kernel_state;
76
77 if (!app_i2c_ensure_mutex())
78 {
79 kernel_state = osKernelGetState();
80 return (kernel_state == osKernelInactive);
81 }
82
83 timeout = (timeout_ms == 0U) ? osWaitForever : timeout_ms;
84 return (osMutexAcquire(s_app_i2c_mutex, timeout) == osOK);
85}

87void app_i2c_unlock(void)
88{
89 if (s_app_i2c_mutex != NULL)
90 {
91 (void)osMutexRelease(s_app_i2c_mutex);
92 }
93}

95HAL_StatusTypeDef app_i2c_master_transmit(I2C_HandleTypeDef *hi2c,
96 uint16_t device_address,
97 const uint8_t *data,
98 uint16_t size,
99 uint32_t timeout_ms)
100{
101 HAL_StatusTypeDef status;
102 uint32_t attempt;
103 uint8_t *tx_data;
104
105 if ((hi2c == NULL) || ((data == NULL) && (size > 0U)))
106 {
107 return HAL_ERROR;
108 }
109 if (!app_i2c_lock(0U))
110 {
111 return HAL_TIMEOUT;
112 }
113
114 tx_data = (uint8_t *)data;
115 status = HAL_ERROR;
116
117 for (attempt = 0U; attempt < APP_I2C_RETRY_COUNT; attempt++)
118 {
119 status = HAL_I2C_Master_Transmit(hi2c,
120 device_address,
121 tx_data,
122 size,
123 app_i2c_normalize_timeout(timeout_ms));
124 if (status == HAL_OK)
125 {
126 break;
127 }
128
129 if (app_i2c_should_recover(hi2c, status))
130 {
131 (void)app_i2c_recover_locked(hi2c);
132 }
133
134 if ((attempt + 1U) < APP_I2C_RETRY_COUNT)
135 {
136 app_delay_ms(APP_I2C_RETRY_DELAY_MS);
137 }
138 }
139
140 app_i2c_unlock();
141 return status;
142}

296static bool app_i2c_ensure_mutex(void)
297{
298 osKernelState_t kernel_state;
299
300 if (s_app_i2c_mutex != NULL)
301 {
302 return true;
303 }
304
305 kernel_state = osKernelGetState();
306 if ((kernel_state == osKernelReady) || (kernel_state == osKernelRunning))
307 {
308 s_app_i2c_mutex = osMutexNew(&s_app_i2c_mutex_attr);
309 }
310
311 return (s_app_i2c_mutex != NULL);
312}

326static bool app_i2c_should_recover(I2C_HandleTypeDef *hi2c, HAL_StatusTypeDef status)
327{
328 uint32_t error_flags;
329
330 if ((hi2c == NULL) || (status == HAL_OK))
331 {
332 return false;
333 }
334
335 if ((status == HAL_BUSY) || (status == HAL_TIMEOUT))
336 {
337 return true;
338 }
339
340 error_flags = HAL_I2C_GetError(hi2c);
341
342 return ((error_flags & (HAL_I2C_ERROR_BERR |
343 HAL_I2C_ERROR_ARLO |
344 HAL_I2C_ERROR_OVR |
345 HAL_I2C_ERROR_DMA |
346 HAL_I2C_ERROR_TIMEOUT |
347 HAL_I2C_ERROR_SIZE |
348 HAL_I2C_ERROR_DMA_PARAM)) != 0U);
349}
350
352static bool app_i2c_recover_locked(I2C_HandleTypeDef *hi2c)
353{
354 if (hi2c == NULL)
355 {
356 return false;
357 }
358
359 (void)HAL_I2C_DeInit(hi2c);
360
361 if (HAL_I2C_Init(hi2c) != HAL_OK)
362 {
363 return false;
364 }
365 if (HAL_I2CEx_ConfigAnalogFilter(hi2c, I2C_ANALOGFILTER_ENABLE) != HAL_OK)
366 {
367 return false;
368 }
369 if (HAL_I2CEx_ConfigDigitalFilter(hi2c, 0U) != HAL_OK)
370 {
371 return false;
372 }
373
374 return true;
375}
