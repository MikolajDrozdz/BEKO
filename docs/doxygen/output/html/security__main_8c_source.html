367void security_main_create_task(void)
368{
369 if (s_security_mutex == NULL)
370 {
371 s_security_mutex = osMutexNew(NULL);
372 if (s_security_mutex == NULL)
373 {
374 printf("SEC: mutex create failed\r\n");
375 return;
376 }
377 }
378
379 if (s_security_cmd_queue == NULL)
380 {
381 s_security_cmd_queue = osMessageQueueNew(SECURITY_CMD_QUEUE_DEPTH, sizeof(security_cmd_t), NULL);
382 if (s_security_cmd_queue == NULL)
383 {
384 printf("SEC: queue create failed\r\n");
385 return;
386 }
387 }
388
389 if (s_security_task == NULL)
390 {
391 memset(&s_security_task_cb, 0, sizeof(s_security_task_cb));
392 memset(s_security_task_stack, 0, sizeof(s_security_task_stack));
393 printf("SEC: task memory cb=%p stack=%p size=%lu\r\n",
394 (void *)&s_security_task_cb,
395 (void *)s_security_task_stack,
396 (unsigned long)sizeof(s_security_task_stack));
397 s_security_task = osThreadNew(security_main_task_fn, NULL, &s_security_task_attr);
398 if (s_security_task == NULL)
399 {
400 printf("SEC: task create failed\r\n");
401 }
402 }
403}

699static bool security_get_frame_keys_mode_internal(uint16_t local_id,
700 uint16_t peer_id,
701 security_frame_key_mode_t mode,
702 const uint8_t *code_in,
703 uint8_t code_in_len,
704 bool use_explicit_code,
705 uint8_t enc_key_out[16],
706 uint8_t hmac_key_out[32])
707{
708 uint8_t base_key[16];
709 uint8_t digest[32];
710 uint8_t info[8];
711 uint8_t code[SECURITY_CODE_MAX];
712 uint8_t code_len = 0U;
713 uint16_t domain_id;
714 bool ok = false;
715
716 if ((local_id == 0U) || (peer_id == 0U) || (enc_key_out == NULL) || (hmac_key_out == NULL))
717 {
718 return false;
719 }
720
721 memset(base_key, 0, sizeof(base_key));
722 memset(code, 0, sizeof(code));
723 switch (mode)
724 {
725 case SECURITY_FRAME_KEY_MODE_PAIR_V1_32:
726 if (use_explicit_code)
727 {
728 if ((code_in == NULL) || (code_in_len == 0U))
729 {
730 return false;
731 }
732 code_len = (code_in_len > SECURITY_CODE_MAX) ? SECURITY_CODE_MAX : code_in_len;
733 memcpy(code, code_in, code_len);
734 }
735 else if (!security_lookup_peer_code(peer_id, code, &code_len))
736 {
737 return false;
738 }
739 security_peer_link_key_derive(local_id, peer_id, code, code_len, base_key);
740 ok = true;
741 break;
742
743 case SECURITY_FRAME_KEY_MODE_SHARED:
744 default:
745 memcpy(base_key, s_shared_frame_root_key, sizeof(base_key));
746 ok = true;
747 break;
748 }
749
750 if (!ok)
751 {
752 laviet_secure_zero(code, sizeof(code));
753 return false;
754 }
755
756 domain_id = (peer_id == LAVIET_BROADCAST_ID) ?
757 LAVIET_BROADCAST_ID :
758 ((local_id < peer_id) ? local_id : peer_id);
759 info[0] = (uint8_t)'L';
760 info[1] = (uint8_t)'V';
761 info[2] = (uint8_t)'1';
762 info[3] = (uint8_t)'K';
763 info[4] = (uint8_t)(domain_id >> 8);
764 info[5] = (uint8_t)domain_id;
765 info[6] = 0U;
766 info[7] = 1U;
767 ok = laviet_hmac_sha256(base_key, sizeof(base_key), info, sizeof(info), digest);
768 if (ok)
769 {
770 memcpy(enc_key_out, digest, 16U);
771 info[7] = 2U;
772 ok = laviet_hmac_sha256(base_key, sizeof(base_key), info, sizeof(info), hmac_key_out);
773 }
774
775 laviet_secure_zero(base_key, sizeof(base_key));
776 laviet_secure_zero(digest, sizeof(digest));
777 laviet_secure_zero(info, sizeof(info));
778 laviet_secure_zero(code, sizeof(code));
779 return ok;
780}

1236static uint8_t security_pack_bits(uint8_t *buf, uint8_t bit_pos, uint32_t value, uint8_t width)
1237{
1238 uint8_t bit;
1239
1240 for (bit = 0U; bit < width; bit++)
1241 {
1242 uint8_t dst_bit = (uint8_t)(bit_pos + bit);
1243 uint8_t dst_byte = (uint8_t)(dst_bit / 8U);
1244 uint8_t dst_mask = (uint8_t)(1U << (dst_bit % 8U));
1245
1246 if (((value >> bit) & 0x01UL) != 0UL)
1247 {
1248 buf[dst_byte] |= dst_mask;
1249 }
1250 else
1251 {
1252 buf[dst_byte] &= (uint8_t)~dst_mask;
1253 }
1254 }
1255
1256 return (uint8_t)(bit_pos + width);
1257}

1260static uint8_t security_unpack_bits(const uint8_t *buf, uint8_t bit_pos, uint8_t width, uint32_t *value_out)
1261{
1262 uint32_t value = 0UL;
1263 uint8_t bit;
1264
1265 if (value_out == NULL)
1266 {
1267 return bit_pos;
1268 }
1269
1270 for (bit = 0U; bit < width; bit++)
1271 {
1272 uint8_t src_bit = (uint8_t)(bit_pos + bit);
1273 uint8_t src_byte = (uint8_t)(src_bit / 8U);
1274 uint8_t src_mask = (uint8_t)(1U << (src_bit % 8U));
1275
1276 if ((buf[src_byte] & src_mask) != 0U)
1277 {
1278 value |= (1UL << bit);
1279 }
1280 }
1281
1282 *value_out = value;
1283 return (uint8_t)(bit_pos + width);

1418static void security_peer_link_key_derive(uint32_t local_node_id,
1419 uint32_t peer_node_id,
1420 const uint8_t *code,
1421 uint8_t code_len,
1422 uint8_t key_out[16])
1423{
1424 static const uint8_t label[] = "SEC:PAIR:V1";
1425 uint8_t info[sizeof(label) - 1U + 8U];
1426 uint8_t digest[LAVIET_SHA256_LEN];
1427 uint32_t lo;
1428 uint32_t hi;
1429
1430 if ((key_out == NULL) || (code == NULL) || (code_len == 0U))
1431 {
1432 return;
1433 }
1434
1435 lo = (local_node_id < peer_node_id) ? local_node_id : peer_node_id;
1436 hi = (local_node_id < peer_node_id) ? peer_node_id : local_node_id;
1437 memset(info, 0, sizeof(info));
1438 memset(digest, 0, sizeof(digest));
1439 memcpy(info, label, sizeof(label) - 1U);
1440 security_be32_write(&info[sizeof(label) - 1U], lo);
1441 security_be32_write(&info[sizeof(label) - 1U + 4U], hi);
1442
1443 if (laviet_hmac_sha256(code, code_len, info, (uint16_t)sizeof(info), digest))
1444 {
1445 memcpy(key_out, digest, 16U);
1446 }
1447 else
1448 {
1449 memset(key_out, 0, 16U);
1450 }
1451
1452 laviet_secure_zero(info, sizeof(info));
1453 laviet_secure_zero(digest, sizeof(digest));
1454}
