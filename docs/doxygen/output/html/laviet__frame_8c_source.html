80uint16_t laviet_local_node_id(void)
81{
82 uint32_t uid0 = HAL_GetUIDw0();
83 uint32_t uid1 = HAL_GetUIDw1();
84 uint32_t uid2 = HAL_GetUIDw2();
85 uint32_t hash = 2166136261UL;
86 uint8_t i;
87 const uint8_t *uid_bytes = (const uint8_t *)&uid0;
88
89 for (i = 0U; i < 4U; i++)
90 {
91 hash ^= uid_bytes[i];
92 hash *= 16777619UL;
93 }
94 uid_bytes = (const uint8_t *)&uid1;
95 for (i = 0U; i < 4U; i++)
96 {
97 hash ^= uid_bytes[i];
98 hash *= 16777619UL;
99 }
100 uid_bytes = (const uint8_t *)&uid2;
101 for (i = 0U; i < 4U; i++)
102 {
103 hash ^= uid_bytes[i];
104 hash *= 16777619UL;
105 }
106
107 hash = (hash ^ (hash >> 16)) & 0xFFFFUL;
108 if ((hash == 0U) || (hash == LAVIET_GATEWAY_ID) || (hash == LAVIET_BROADCAST_ID))
109 {
110 hash = (uint32_t)(0x1000U | (hash & 0x0FFFU));
111 if ((hash == 0U) || (hash == LAVIET_GATEWAY_ID) || (hash == LAVIET_BROADCAST_ID))
112 {
113 hash = 0x1002U;
114 }
115 }
116
117 return (uint16_t)hash;
118}

120laviet_frame_status_t laviet_frame_validate_plain(const laviet_frame_t *frame)
121{
122 laviet_frame_type_t type;
123 bool broadcast;
124
125 if (frame == NULL)
126 {
127 return LAVIET_STATUS_NULL;
128 }
129
130 if (laviet_frame_version_from_ver_type(frame->ver_type) != LAVIET_FRAME_VERSION)
131 {
132 return LAVIET_STATUS_VERSION;
133 }
134
135 type = laviet_frame_type(frame);
136 if (!laviet_frame_type_known(type))
137 {
138 return LAVIET_STATUS_TYPE;
139 }
140
141 if ((frame->src_id == 0U) ||
142 (frame->src_id == LAVIET_BROADCAST_ID) ||
143 (frame->dst_id == 0U) ||
144 (frame->msg_id == 0U))
145 {
146 return LAVIET_STATUS_ID;
147 }
148
149 if (frame->payload_len > LAVIET_MAX_PAYLOAD)
150 {
151 return LAVIET_STATUS_PAYLOAD;
152 }
153
154 broadcast = (frame->dst_id == LAVIET_BROADCAST_ID);
155 if (broadcast)
156 {
157 if (((frame->flags & LAVIET_FLAG_BROADCAST) == 0U) ||
158 ((frame->flags & LAVIET_FLAG_ACK_REQUIRED) != 0U))
159 {
160 return LAVIET_STATUS_FLAGS;
161 }
162 if ((frame->src_id != LAVIET_GATEWAY_ID) &&
163 (type != LAVIET_TYPE_DATA) &&
164 (type != LAVIET_TYPE_RESP) &&
165 (type != LAVIET_TYPE_PAIR_REQ))
166 {
167 return LAVIET_STATUS_FLAGS;
168 }
169 }
170 else if ((frame->flags & LAVIET_FLAG_BROADCAST) != 0U)
171 {
172 return LAVIET_STATUS_FLAGS;
173 }
174
175 switch (type)
176 {
177 case LAVIET_TYPE_DATA:
178 if ((frame->payload_len == 0U) || (frame->payload_len > LAVIET_MAX_PAYLOAD))
179 {
180 return LAVIET_STATUS_PAYLOAD;
181 }
182 if (((frame->flags & LAVIET_FLAG_IS_ACK) != 0U) ||
183 ((frame->flags & LAVIET_FLAG_PAIRING) != 0U) ||
184 ((frame->flags & LAVIET_FLAG_CONFIG_ACCESS) != 0U) ||
185 ((frame->flags & LAVIET_FLAG_COUNTER_OVERRIDE) != 0U) ||
186 ((frame->flags & LAVIET_FLAG_KEY_UPDATE) != 0U))
187 {
188 return LAVIET_STATUS_FLAGS;
189 }
190 break;
191
192 case LAVIET_TYPE_ACK:
193 if (frame->payload_len != LAVIET_ACK_PAYLOAD_LEN)
194 {
195 return LAVIET_STATUS_PAYLOAD;
196 }
197 if (((frame->flags & LAVIET_FLAG_IS_ACK) == 0U) ||
198 ((frame->flags & LAVIET_FLAG_ACK_REQUIRED) != 0U) ||
199 ((frame->flags & LAVIET_FLAG_ENCRYPTED) != 0U))
200 {
201 return LAVIET_STATUS_FLAGS;
202 }
203 break;
204
205 case LAVIET_TYPE_RESP:
206 if ((frame->flags & (LAVIET_FLAG_IS_ACK |
207 LAVIET_FLAG_PAIRING |
208 LAVIET_FLAG_CONFIG_ACCESS |
209 LAVIET_FLAG_COUNTER_OVERRIDE |
210 LAVIET_FLAG_KEY_UPDATE)) != 0U)
211 {
212 return LAVIET_STATUS_FLAGS;
213 }
214 break;
215
216 case LAVIET_TYPE_PAIR_REQ:
217 case LAVIET_TYPE_PAIR_RESP:
218 if (frame->payload_len != LAVIET_PAIR_PAYLOAD_LEN)
219 {
220 return LAVIET_STATUS_PAYLOAD;
221 }
222 if (((frame->flags & LAVIET_FLAG_PAIRING) == 0U) ||
223 ((frame->flags & LAVIET_FLAG_CONFIG_ACCESS) != 0U) ||
224 ((frame->flags & LAVIET_FLAG_COUNTER_OVERRIDE) != 0U) ||
225 ((frame->flags & LAVIET_FLAG_KEY_UPDATE) != 0U))
226 {
227 return LAVIET_STATUS_FLAGS;
228 }
229 break;
230
231 case LAVIET_TYPE_CFG:
232 if ((frame->payload_len < 2U) || (frame->payload_len > LAVIET_MAX_PAYLOAD))
233 {
234 return LAVIET_STATUS_PAYLOAD;
235 }
236 if (((frame->flags & LAVIET_FLAG_CONFIG_ACCESS) == 0U) ||
237 ((frame->flags & LAVIET_FLAG_IS_ACK) != 0U) ||
238 ((frame->flags & LAVIET_FLAG_PAIRING) != 0U))
239 {
240 return LAVIET_STATUS_FLAGS;
241 }
242 break;
243
244 case LAVIET_TYPE_COUNTER_SYNC:
245 if (frame->payload_len != LAVIET_COUNTER_SYNC_PAYLOAD_LEN)
246 {
247 return LAVIET_STATUS_PAYLOAD;
248 }
249 if (((frame->flags & LAVIET_FLAG_COUNTER_OVERRIDE) == 0U) ||
250 ((frame->flags & LAVIET_FLAG_IS_ACK) != 0U) ||
251 (frame->src_id != LAVIET_GATEWAY_ID))
252 {
253 return LAVIET_STATUS_FLAGS;
254 }
255 break;
256
257 case LAVIET_TYPE_KEY_ROTATE:
258 if ((frame->payload_len == 0U) || (frame->payload_len > LAVIET_MAX_PAYLOAD))
259 {
260 return LAVIET_STATUS_PAYLOAD;
261 }
262 if (((frame->flags & LAVIET_FLAG_KEY_UPDATE) == 0U) ||
263 ((frame->flags & LAVIET_FLAG_IS_ACK) != 0U))
264 {
265 return LAVIET_STATUS_FLAGS;
266 }
267 break;
268
269 case LAVIET_TYPE_ERROR:
270 if (frame->payload_len != LAVIET_ERROR_PAYLOAD_LEN)
271 {
272 return LAVIET_STATUS_PAYLOAD;
273 }
274 if ((frame->flags & LAVIET_FLAG_IS_ACK) != 0U)
275 {
276 return LAVIET_STATUS_FLAGS;
277 }
278 break;
279
280 default:
281 return LAVIET_STATUS_TYPE;
282 }
283
284 return LAVIET_STATUS_OK;
285}

287laviet_frame_status_t laviet_frame_encode(const laviet_frame_t *frame,
288 uint8_t *out,
289 uint8_t out_capacity,
290 uint8_t *out_len)
291{
292 laviet_frame_status_t status;
293 uint8_t total_len;
294
295 if ((frame == NULL) || (out == NULL) || (out_len == NULL))
296 {
297 return LAVIET_STATUS_NULL;
298 }
299
300 status = laviet_frame_validate_plain(frame);
301 if (status != LAVIET_STATUS_OK)
302 {
303 return status;
304 }
305
306 total_len = (uint8_t)(LAVIET_FRAME_HEADER_LEN + frame->payload_len + LAVIET_MAC_TAG_LEN);
307 if (out_capacity < total_len)
308 {
309 return LAVIET_STATUS_LENGTH;
310 }
311
312 out[0] = frame->ver_type;
313 out[1] = frame->flags;
314 laviet_be16_write(&out[2], frame->src_id);
315 laviet_be16_write(&out[4], frame->dst_id);
316 laviet_be16_write(&out[6], frame->msg_id);
317 laviet_be32_write(&out[8], frame->counter);
318 out[12] = frame->payload_len;
319 if (frame->payload_len > 0U)
320 {
321 memcpy(&out[LAVIET_FRAME_HEADER_LEN], frame->payload, frame->payload_len);
322 }
323 memcpy(&out[LAVIET_FRAME_HEADER_LEN + frame->payload_len], frame->mac_tag, LAVIET_MAC_TAG_LEN);
324 *out_len = total_len;
325 return LAVIET_STATUS_OK;
326}

328laviet_frame_status_t laviet_frame_decode(const uint8_t *in,
329 uint8_t in_len,
330 laviet_frame_t *frame_out)
331{
332 uint8_t payload_len;
333 uint8_t expected_len;
334 laviet_frame_status_t status;
335
336 if ((in == NULL) || (frame_out == NULL))
337 {
338 return LAVIET_STATUS_NULL;
339 }
340 if ((in_len < LAVIET_FRAME_MIN_LEN) || (in_len > LAVIET_FRAME_MAX_LEN))
341 {
342 return LAVIET_STATUS_LENGTH;
343 }
344
345 payload_len = in[12];
346 if (payload_len > LAVIET_MAX_PAYLOAD)
347 {
348 return LAVIET_STATUS_PAYLOAD;
349 }
350 expected_len = (uint8_t)(LAVIET_FRAME_HEADER_LEN + payload_len + LAVIET_MAC_TAG_LEN);
351 if (expected_len != in_len)
352 {
353 return LAVIET_STATUS_LENGTH;
354 }
355
356 memset(frame_out, 0, sizeof(*frame_out));
357 frame_out->ver_type = in[0];
358 frame_out->flags = in[1];
359 frame_out->src_id = laviet_be16_read(&in[2]);
360 frame_out->dst_id = laviet_be16_read(&in[4]);
361 frame_out->msg_id = laviet_be16_read(&in[6]);
362 frame_out->counter = laviet_be32_read(&in[8]);
363 frame_out->payload_len = payload_len;
364 if (payload_len > 0U)
365 {
366 memcpy(frame_out->payload, &in[LAVIET_FRAME_HEADER_LEN], payload_len);
367 }
368 memcpy(frame_out->mac_tag, &in[LAVIET_FRAME_HEADER_LEN + payload_len], LAVIET_MAC_TAG_LEN);
369
370 status = laviet_frame_validate_plain(frame_out);
371 if (status != LAVIET_STATUS_OK)
372 {
373 memset(frame_out, 0, sizeof(*frame_out));
374 }
375 return status;
376}

378bool laviet_frame_build_mac_input(const laviet_frame_t *frame,
379 uint8_t *out,
380 uint8_t out_capacity,
381 uint8_t *out_len)
382{
383 uint8_t len;
384
385 if ((frame == NULL) || (out == NULL) || (out_len == NULL))
386 {
387 return false;
388 }

390 {
391 return false;
392 }
393
394 len = (uint8_t)(LAVIET_FRAME_HEADER_LEN + frame->payload_len);
395 if (out_capacity < len)
396 {
397 return false;
398 }
399
400 out[0] = frame->ver_type;
401 out[1] = frame->flags;
402 laviet_be16_write(&out[2], frame->src_id);
403 laviet_be16_write(&out[4], frame->dst_id);
404 laviet_be16_write(&out[6], frame->msg_id);
405 laviet_be32_write(&out[8], frame->counter);
406 out[12] = frame->payload_len;
407 if (frame->payload_len > 0U)
408 {
409 memcpy(&out[LAVIET_FRAME_HEADER_LEN], frame->payload, frame->payload_len);
410 }
411 *out_len = len;
412 return true;
413}
